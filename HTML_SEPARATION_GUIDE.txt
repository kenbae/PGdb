# ============================================================
# 설치 방법
# ============================================================

# 1. 필수 패키지 설치
pip install fastapi uvicorn[standard] jinja2 sqlalchemy psycopg2-binary pyyaml pandas

# 또는
pip install -r requirements_fastapi.txt


# 2. 폴더 생성
mkdir templates
mkdir static
mkdir static/css
mkdir static/js
mkdir static/images


# 3. 파일 배치
# - dashboard_api_v2.py를 프로젝트 루트에 배치
# - index.html을 templates/ 폴더에 배치


# ============================================================
# 실행 방법
# ============================================================

# 기본 실행
python dashboard_api_v2.py

# 포트 변경
python dashboard_api_v2.py --port 8080

# 개발 모드 (자동 재로드)
python dashboard_api_v2.py --reload

# 접속
# http://localhost:8001


# ============================================================
# 장점
# ============================================================

✅ Python 코드와 HTML 완전 분리
   - dashboard_api_v2.py: API 로직만
   - index.html: UI와 프론트엔드 로직만

✅ 유지보수 용이
   - HTML 수정 시 Python 파일 건드릴 필요 없음
   - 디자이너가 HTML만 수정 가능

✅ 확장 가능
   - 여러 페이지 추가 가능 (about.html, settings.html 등)
   - 정적 파일(CSS, JS, 이미지) 별도 관리

✅ 템플릿 엔진 활용
   - Jinja2로 동적 데이터 주입 가능
   - 코드 재사용성 향상


# ============================================================
# HTML 수정 방법
# ============================================================

templates/index.html 파일만 수정하면 됩니다:

1. 디자인 변경
   - <style> 태그 내부의 CSS 수정
   - Tailwind CSS 클래스 변경

2. 레이아웃 변경
   - HTML 구조 수정
   - 차트 위치 변경

3. 기능 추가
   - <script> 태그 내부의 JavaScript 수정
   - 새로운 API 엔드포인트 호출

# ============================================================
# Jinja2 템플릿 기능 활용
# ============================================================

# dashboard_api_v2.py 수정
@app.get("/", response_class=HTMLResponse)
async def root(request: Request):
    # 서버에서 데이터 전달
    data = {
        "request": request,
        "app_name": "Crypto Trading Dashboard",
        "version": "1.0.0",
        "user": "Admin"
    }
    return templates.TemplateResponse("index.html", data)

# templates/index.html에서 사용
<h1>{{ app_name }}</h1>
<p>Version: {{ version }}</p>
<p>User: {{ user }}</p>
